macro_rules! tr {
    ($l:expr, $hans:literal, $hant:literal, $en:literal $(,)?) => {
        match $l {
            $crate::lang::Lang::Hans => format!($hans),
            $crate::lang::Lang::Hant => format!($hant),
            $crate::lang::Lang::En => format!($en),
        }
    };
}
