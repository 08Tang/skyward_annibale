81 /// call it formattable, and where the table has it encrypted whole, to say
82 /// where its key is kept as well, for the system makes the key anew with
83 /// the filesystem.
84 pub fn made_afresh(entry: &Entry) -> bool {
85     let formattable = entry.extras.iter().any(|e| e == "formattable");
86     let key_kept = entry.extra("metadata_encryption").is_none() || entry.extra("keydirectory").is_some();
87     formattable && key_kept
88 }

90 /// Where the phone keeps the key for the filesystem an entry describes.
91 pub fn key_dir(entry: &Entry) -> Result<PathBuf> {
92     let dir = entry.extra("keydirectory").context("the phone says where the key for its data partition is not")?;
93     Ok(Path::new(dir).join(KEY_DIR))
94 }
