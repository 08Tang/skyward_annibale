166     partition("system_dlkm", MOUNT, Screen::Mount),
167     partition("odm_dlkm", MOUNT, Screen::Mount),

170 /// The entries `query` finds, best first: a title in `l` holding it, then a
171 /// key starting with it, then a key holding it. The keys are those of `l`
172 /// and the English ones.
173 pub fn find(query: &str, l: Lang) -> Vec<&'static Entry> {
174     let q = typed(query.trim());
175     if q.is_empty() {
176         return Vec::new();
177     }
178     let rank = |e: &Entry| {
179         let keys = || l.pick(e.keys).split(' ').chain(Lang::En.pick(e.keys).split(' '));
180         if typed(l.pick(e.title)).contains(&q) {
181             Some(0)
182         } else if keys().any(|k| k.starts_with(&q)) {
183             Some(1)
184         } else if keys().any(|k| k.contains(&q)) {
185             Some(2)
186         } else {
187             None
188         }
189     };
190     let mut found: Vec<(u8, usize, &Entry)> = ENTRIES.iter().enumerate().filter_map(|(k, e)| rank(e).map(|r| (r, k, e))).collect();
191     found.sort_by_key(|&(r, k, _)| (r, k));
192     found.into_iter().map(|(_, _, e)| e).collect()
193 }

195 /// What [`find`] finds that the phone has, as `offers` tells of each page:
196 /// a phone whose WLAN the recovery does not bring up has no WLAN page.
197 pub fn find_on(query: &str, l: Lang, offers: impl Fn(Screen) -> bool) -> Vec<&'static Entry> {
198     find(query, l).into_iter().filter(|e| offers(e.screen)).collect()
199 }
